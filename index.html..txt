<script>
  let tasks = JSON.parse(localStorage.getItem("tasks") || "[]");

  function renderTasks() {
    document.getElementById("tasks").innerHTML = "";
    tasks.forEach((task, index) => {
      const li = document.createElement("li");
      li.append(document.createTextNode(task));
      const del = document.createElement("button");
      del.textContent = "Delete";
      del.onclick = () => {
        tasks.splice(index, 1);
        localStorage.setItem("tasks", JSON.stringify(tasks));
        renderTasks();
      };
      li.appendChild(del);
      document.getElementById("tasks").appendChild(li);
    });
  }

  function addTask() {
    const input = document.getElementById("task");
    if (!input.value.trim()) return;
    tasks.push(input.value.trim());
    localStorage.setItem("tasks", JSON.stringify(tasks));
    input.value = "";
    renderTasks();
  }

  function saveNote() {
    localStorage.setItem(
      "studyNote", document.getElementById("note").value
    );
    document.getElementById("status").textContent = "Notes saved!";
  }

  document.getElementById("note").value =
    localStorage.getItem("studyNote") || "";

  const quotes = [
    "Keep learning. Keep growing.",
    "Every expert was once a beginner.",
    "Your future starts with what you learn today."
  ];

  function newQuote() {
    document.getElementById("quote").textContent =
      quotes[Math.floor(Math.random() * quotes.length)];
  }

  renderTasks();
</script>
